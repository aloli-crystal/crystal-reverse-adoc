= Usage

== As a Library

[source,crystal]
----
require "reverse_adoc"

html = "<h1>Title</h1><p>A paragraph with <strong>bold</strong> text.</p>"
adoc = ReverseAdoc.convert(html)
puts adoc
----

== Word .docx Support

[source,crystal]
----
require "reverse_adoc"

adoc = ReverseAdoc.convert_docx("document.docx")
puts adoc
----

== CLI

[source,shell]
----
reverse-adoc input.html
reverse-adoc input.html -o output.adoc
reverse-adoc document.docx
reverse-adoc -v
reverse-adoc -h
----
